        }
    }
    const FINAL_QUESTIONS = uniqueBank.slice(0, 1050); // final pool

    let currentFilter = "all";
    let filteredIndices = [];
    let currentQIndex = 0;
    let answeredCorrect = [];
    let userSelected = [];
    let totalScore = 0;

    function buildFilteredIndices() {
        if (currentFilter === "all") {
            filteredIndices = FINAL_QUESTIONS.map((_, idx) => idx);
        } else {
            filteredIndices = FINAL_QUESTIONS.reduce((acc, q, idx) => {
                if (q.tag === currentFilter) acc.push(idx);
                return acc;
            }, []);
        }
        // reset progress for new filter
        answeredCorrect = new Array(filteredIndices.length).fill(false);
        userSelected = new Array(filteredIndices.length).fill(null);
        totalScore = 0;
        currentQIndex = 0;
        saveScoreUI();
        updateCounter();
        renderCurrent();
    }

    function getCurrentQuestion() {
        return FINAL_QUESTIONS[filteredIndices[currentQIndex]];
    }

    function saveScoreUI() {
        document.getElementById("scoreSpan").innerHTML = `⭐ Score: ${totalScore} / ${filteredIndices.length}`;
    }

    function updateCounter() {
        document.getElementById("qCount").innerHTML = `Q${currentQIndex+1} / ${filteredIndices.length}`;
    }
