        document.querySelectorAll('.option').forEach(opt => {
            opt.addEventListener('click', () => {
                if (correctStatus[currentQ]) {
                    const fbDiv = document.getElementById("fbBox");
                    document.getElementById("fbText").innerHTML = `🔒 Already answered correctly! Answer: ${q.correct}`;
                    fbDiv.classList.add("show");
                    return;
                }
                const selected = opt.getAttribute('data-optval');
                const isCor = (selected === q.correct);
                const fbDiv = document.getElementById("fbBox");
                const fbSpan = document.getElementById("fbText");
                if (isCor) {
                    if (!correctStatus[currentQ]) {
                        correctStatus[currentQ] = true;
                        totalScore++;
                        saveScoreUI();
                        userAttempt[currentQ] = selected;
                        fbSpan.innerHTML = `🎉 Correct! ${q.correct}. ${getExplain(q)}`;
                        fbDiv.classList.add("show");
                        const nextB = document.getElementById("nextBtn");
                        if (nextB) nextB.disabled = false;
                    } else {
                        fbSpan.innerHTML = `ℹ️ Already correct: ${q.correct}`;
                        fbDiv.classList.add("show");
                    }
                } else {
                    userAttempt[currentQ] = selected;
                    fbSpan.innerHTML = `❌ Wrong. The correct answer is: ${q.correct}. ${getExplain(q)}`;
                    fbDiv.classList.add("show");
                    const nextB = document.getElementById("nextBtn");
                    if (nextB) nextB.disabled = true;
                }
            });
        });
        const nxt = document.getElementById("nextBtn");
        if (nxt) {
            nxt.addEventListener('click', () => {
                if (!correctStatus[currentQ]) {
                    const fbD = document.getElementById("fbBox");
                    document.getElementById("fbText").innerHTML = `⚠️ Please select correct answer before moving. Correct: ${q.correct}`;
                    fbD.classList.add("show");
                    return;
                }
                if (currentQ + 1 < filteredIndices.length) {
                    currentQ++;
                    renderCurrent();
                    updateCounter();
                } else {
                    showCompletion();
                }
            });
        }
    }
    
    function getExplain(q) {
        if (q.correct.includes("Proton")) return "🔬 Protons have positive charge in nucleus.";
        if (q.correct.includes("Mitochondria")) return "⚡ Mitochondria produce ATP via cellular respiration.";
        if (q.correct.includes("Photon")) return "📡 Photons are massless particles of light.";
        if (q.correct.includes("Gamma")) return "☢️ Gamma rays are electromagnetic with no mass.";
        return "📘 Explanation based on standard science curriculum.";
    }
    
    function showCompletion() {
        const container = document.getElementById("quizDynamic");
        container.innerHTML = `
            <div class="completion">
                <h2>🔬 Quiz Completed! 🎉</h2>
                <p style="font-size:1.4rem; margin:1rem;">Final Score: ${totalScore} / ${filteredIndices.length}</p>
                <p>✅ You've mastered Patwari General Science: Atomic · Biology · Energy · Computer · Space · Quantum</p>
                <button class="reset-btn" id="finalRestart" style="background:#2a7f6b;">⟳ Start Over</button>
            </div>
        `;
        document.getElementById("finalRestart")?.addEventListener('click', () => resetAll());
    }
    
    function resetAll() {
        currentQ = 0;
        correctStatus.fill(false);
        userAttempt.fill(null);
        totalScore = 0;
        saveScoreUI();
        updateCounter();
        renderCurrent();
    }
    
    function escapeHtml(str) {
        return str.replace(/[&<>]/g, function(m) {
            if (m === '&') return '&amp;'; if (m === '<') return '&lt;'; if (m === '>') return '&gt;';
            return m;
        });
    }
    
    document.getElementById("hardReset").addEventListener('click', resetAll);
    document.querySelectorAll(".filter-btn").forEach(btn => {
        btn.addEventListener('click', () => {
            document.querySelectorAll(".filter-btn").forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            currentFilter = btn.getAttribute('data-filter');
            buildFiltered();
        });
    });
    buildFiltered();
